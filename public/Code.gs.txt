const PARENT_FOLDER_ID = "1krBytez3bNzfFe-NQA6MdJxTPBMKYN5Q"; // Thay ID thu muc Google Drive
const SPREADSHEET_ID = "1ZXbqdF3yxuXBZjNgW4x7O7nl7AhgekYkZExcg5pz8IE";     // Thay ID Google Sheet
const SHEET_NAME = "Database";                      // Doi ten Sheet neu can

// GET: Dong bo trang thai tat ca villa tu Google Sheet ve cho cac thiet bi
function doGet(e) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];
    const rows = sheet.getDataRange().getValues();

    const data = {};
    // rows[0] la Header: [Mã Villa, Ảnh Trước (URL), Thời Gian Trước, Ảnh Sau (URL), Thời Gian Sau, Trạng Thái]
    for (let i = 1; i < rows.length; i++) {
      const row = rows[i];
      const villaId = String(row[0]).trim();
      if (!villaId) continue;

      const numMatch = villaId.match(/\d+/);
      const num = numMatch ? parseInt(numMatch[0], 10) : null;
      if (!num) continue;

      data[num] = {
        before: row[1] ? String(row[1]) : null,
        beforeTime: row[2] ? String(row[2]) : null,
        after: row[3] ? String(row[3]) : null,
        afterTime: row[4] ? String(row[4]) : null,
        status: row[5] ? String(row[5]) : "Chưa hoàn thành"
      };
    }

    return ContentService.createTextOutput(
      JSON.stringify({
        status: "success",
        total: Object.keys(data).length,
        villas: data
      })
    ).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(
      JSON.stringify({
        status: "error",
        message: err.message
      })
    ).setMimeType(ContentService.MimeType.JSON);
  }
}

// POST: Nhan anh moi, upload Drive va ghi de vao Google Sheet
function doPost(e) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);

    const data = JSON.parse(e.postData.contents);
    const villaId = String(data.villaId || "").trim(); // "Villa 01" hoac "Villa 1" hoac "1"
    const type = String(data.type || "").toLowerCase().trim(); // "before" | "after"
    const rawImage = data.image;
    const mimeType = data.mimeType || "image/jpeg";

    if (!villaId || !type || !rawImage) {
      throw new Error("Missing required fields: villaId, type, image");
    }

    // Lay so thu tu Villa (vd: "Villa 01" -> "Villa 1")
    const numMatch = villaId.match(/\d+/);
    const villaNum = numMatch ? parseInt(numMatch[0], 10) : villaId;
    const typeLabel = type === "before" ? "Before" : "After";

    // 1. Luu tat ca anh vao cung 1 thu muc me PARENT_FOLDER_ID (khong chia thu muc con)
    let targetFolder;
    if (PARENT_FOLDER_ID && PARENT_FOLDER_ID !== "YOUR_PARENT_FOLDER_ID_HERE") {
      targetFolder = DriveApp.getFolderById(PARENT_FOLDER_ID);
    } else {
      targetFolder = DriveApp.getRootFolder();
    }

    // 2. Decode Base64 & Tao file anh voi ten format: "Villa X Before.jpg" hoac "Villa X After.jpg"
    const base64Data = rawImage.replace(/^data:image\/\w+;base64,/, "");
    const decodedBlob = Utilities.newBlob(
      Utilities.base64Decode(base64Data),
      mimeType
    );

    const fileName = `Villa ${villaNum} ${typeLabel}.jpg`;

    // Neu da co file cu cung ten trong folder thi xoa hoac ghi de de khong bi trung
    const existingFiles = targetFolder.getFilesByName(fileName);
    while (existingFiles.hasNext()) {
      const oldFile = existingFiles.next();
      oldFile.setTrashed(true);
    }

    decodedBlob.setName(fileName);
    const createdFile = targetFolder.createFile(decodedBlob);
    createdFile.setSharing(
      DriveApp.Access.ANYONE_WITH_LINK,
      DriveApp.Permission.VIEW
    );
    const fileUrl = createdFile.getUrl();

    // 3. Ghi log & Cap nhat vao Google Sheet
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];

    // Tao Header neu trang tinh con moi
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Mã Villa",
        "Ảnh Trước (URL)",
        "Thời Gian Trước",
        "Ảnh Sau (URL)",
        "Thời Gian Sau",
        "Trạng Thái"
      ]);
      sheet.getRange(1, 1, 1, 6).setFontWeight("bold").setBackground("#f3f4f6");
    }

    const standardVillaId = `Villa ${String(villaNum).padStart(2, '0')}`;
    const now = new Date();
    const timeDisplay = Utilities.formatDate(
      now,
      Session.getScriptTimeZone() || "GMT+7",
      "yyyy-MM-dd HH:mm:ss"
    );

    const rows = sheet.getDataRange().getValues();
    let targetRowIndex = -1;

    for (let i = 1; i < rows.length; i++) {
      const cellVal = String(rows[i][0]).trim();
      const rowNumMatch = cellVal.match(/\d+/);
      const rowNum = rowNumMatch ? parseInt(rowNumMatch[0], 10) : null;
      if (cellVal === standardVillaId || (rowNum !== null && rowNum === villaNum)) {
        targetRowIndex = i + 1;
        break;
      }
    }

    if (targetRowIndex === -1) {
      // Them dong moi neu chua co
      const newRow = [
        standardVillaId,
        type === "before" ? fileUrl : "",
        type === "before" ? timeDisplay : "",
        type === "after" ? fileUrl : "",
        type === "after" ? timeDisplay : "",
        "Chưa hoàn thành"
      ];
      sheet.appendRow(newRow);
      targetRowIndex = sheet.getLastRow();
    } else {
      // Cap nhat URL anh va thoi gian
      if (type === "before") {
        sheet.getRange(targetRowIndex, 2).setValue(fileUrl);
        sheet.getRange(targetRowIndex, 3).setValue(timeDisplay);
      } else if (type === "after") {
        sheet.getRange(targetRowIndex, 4).setValue(fileUrl);
        sheet.getRange(targetRowIndex, 5).setValue(timeDisplay);
      }
    }

    // Danh gia trang thai hoan thanh
    const beforeUrlVal = sheet.getRange(targetRowIndex, 2).getValue();
    const afterUrlVal = sheet.getRange(targetRowIndex, 4).getValue();
    const isCompleted = Boolean(beforeUrlVal && afterUrlVal);

    sheet.getRange(targetRowIndex, 6).setValue(isCompleted ? "Hoàn Thành" : "Chưa hoàn thành");

    return ContentService.createTextOutput(
      JSON.stringify({
        status: "success",
        fileUrl: fileUrl,
        fileName: fileName,
        villaId: standardVillaId,
        type: type,
        isCompleted: isCompleted
      })
    ).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(
      JSON.stringify({
        status: "error",
        message: err.message
      })
    ).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
